SQL Schema (Run once in Supabase SQL Editor — No Auth / Sign-in required!) 
-- 1. Create expenses table for husband and wife
create table if not exists expenses (
  id uuid primary key default gen_random_uuid(),
  owner text not null check (owner in ('husband', 'wife')),
  amount numeric(10, 2) not null,
  category text not null,
  description text not null,
  spent_at date not null default current_date,
  created_at timestamptz default now()
);

-- 2. Create shared recurring household bills table
create table if not exists bills (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  amount numeric(10, 2) not null,
  due_day integer check (due_day between 1 and 31),
  category text default 'Utilities',
  payer text default 'Split 50/50',
  is_autopay boolean default false,
  status text default 'unpaid' check (status in ('unpaid', 'paid')),
  created_at timestamptz default now()
);

-- 3. Enable public anon access (NO LOGIN REQUIRED)
alter table expenses enable row level security;
alter table bills enable row level security;

create policy "Public expenses access" on expenses for all to anon using (true) with check (true);
create policy "Public bills access" on bills for all to anon using (true) with check (true);
